Блок кода

data-type="code"

Open →
<section class="slide" data-type="code">
  <div class="slide-header"><h2>PyTorch implementation</h2><span class="slide-kicker">Python</span></div>
  <div class="code-block">
    <div class="code-header">
      <span class="code-lang">Python · train_step.py</span>
    </div>
    <pre><code class="language-python">import torch

def train_step(model, batch, optimizer, loss_fn):
    optimizer.zero_grad()
    preds = model(batch.x)
    loss = loss_fn(preds, batch.y)
    loss.backward()
    optimizer.step()
    return loss.item()</code></pre>
  </div>
</section>
