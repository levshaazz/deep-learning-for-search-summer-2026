from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

name = "bert-base-uncased"
tokenizer = AutoTokenizer.from_pretrained(name)             # WordPiece
model = AutoModelForSequenceClassification.from_pretrained(name, num_labels=2)

text = "This movie was absolutely wonderful!"
inputs = tokenizer(text, return_tensors="pt",
                   truncation=True, max_length=512)         # BERT context = 512

with torch.no_grad():
    logits = model(**inputs).logits                         # uses the [CLS] head
    probs = torch.softmax(logits, dim=-1)

pred = probs.argmax(-1).item()                              # 0 / 1
